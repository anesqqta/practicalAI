+    cout << "Не пройдено: " << totalTests - passedTests << endl;
+
+    if (passedTests == totalTests) {
+        cout << "\nУсі регресійні тести пройдено успішно!" << endl;
+        return 0;
+    }
+
+    cout << "\nВиявлено помилки під час тестування!" << endl;
